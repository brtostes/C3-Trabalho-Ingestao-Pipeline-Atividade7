import boto3
import csv
import io
import json
import os
import urllib.parse
from datetime import datetime, timezone

s3 = boto3.client("s3")
sqs = boto3.client("sqs")

QUEUE_URL = os.environ["QUEUE_URL"]


def detectar_delimitador(texto):
    primeira_linha = texto.splitlines()[0] if texto.splitlines() else ""

    candidatos = [",", ";", "|", "\t"]

    contagens = {
        delimitador: primeira_linha.count(delimitador)
        for delimitador in candidatos
    }

    return max(contagens, key=contagens.get)


def lambda_handler(event, context):

    total_arquivos = 0
    total_mensagens = 0

    for record in event.get("Records", []):

        bucket = record["s3"]["bucket"]["name"]

        key = urllib.parse.unquote_plus(
            record["s3"]["object"]["key"]
        )

        print(
            json.dumps(
                {
                    "evento": "arquivo_recebido",
                    "bucket": bucket,
                    "key": key
                },
                ensure_ascii=False
            )
        )

        objeto = s3.get_object(
            Bucket=bucket,
            Key=key
        )

        conteudo_bytes = objeto["Body"].read()

        # Tenta UTF-8 primeiro e Latin-1 como contingência.
        try:
            texto = conteudo_bytes.decode("utf-8-sig")
            encoding = "utf-8-sig"
        except UnicodeDecodeError:
            texto = conteudo_bytes.decode("latin-1")
            encoding = "latin-1"

        delimitador = detectar_delimitador(texto)

        leitor = csv.DictReader(
            io.StringIO(texto),
            delimiter=delimitador
        )

        mensagens_arquivo = 0

        for numero_linha, linha in enumerate(leitor, start=2):

            payload = {
                str(chave).strip(): (
                    valor.strip()
                    if isinstance(valor, str)
                    else valor
                )
                for chave, valor in linha.items()
                if chave is not None
            }

            mensagem = {
                "tipo": "evento",
                "source_bucket": bucket,
                "source_key": key,
                "row_number": numero_linha,
                "encoding_detectado": encoding,
                "delimitador_detectado": delimitador,
                "payload": payload,
                "produzido_em": datetime.now(
                    timezone.utc
                ).isoformat()
            }

            resposta = sqs.send_message(
                QueueUrl=QUEUE_URL,
                MessageBody=json.dumps(
                    mensagem,
                    ensure_ascii=False
                )
            )

            mensagens_arquivo += 1
            total_mensagens += 1

            print(
                json.dumps(
                    {
                        "evento": "mensagem_enviada",
                        "row_number": numero_linha,
                        "message_id": resposta["MessageId"]
                    },
                    ensure_ascii=False
                )
            )

        total_arquivos += 1

        print(
            json.dumps(
                {
                    "evento": "arquivo_processado",
                    "bucket": bucket,
                    "key": key,
                    "mensagens_enviadas": mensagens_arquivo
                },
                ensure_ascii=False
            )
        )

    return {
        "statusCode": 200,
        "arquivos_processados": total_arquivos,
        "mensagens_enviadas": total_mensagens
    }
